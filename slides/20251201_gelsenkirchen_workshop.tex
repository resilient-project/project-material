\begin{frame}{RESILIENT Projektübersicht}
	\footnotesize
	\begin{columns}
		\begin{column}{0.75\textwidth}
			\begin{itemize}
				\item \alert{Problemdiagnose}: Welt befindet sich in einer Phase der \alert{Unsicherheit} und des \alert{Wandels}
				      (Geopolitik, Energiewende, Klimawandel); Szenarien mit einzelnen deterministischen
				      Lösungen nicht mehr bedarfsgerecht
				\item \alert{Projektziel}: Entwicklung und Demonstration von Werkzeugen für eine resiliente Planung der Energiewende für Industrie, E-fuels und Netze
				\item Baut auf dem schon etablierten, quelloffenen und weltweit genutzten Multi-Vektor-Energiesystemmodell \alert{Python for Power System Analysis (PyPSA)}
				\item \alert{Methodischer Schwerpunkt}: Optimierung unter Unsicherheit (stochastisch und robust),  Dekompositionsmethoden und Heuristiken für Resilienz
				\item \alert{Fallstudien}: \alert{Industriewende in NRW}, Netzplanung in BW, E-Fuels in Skandinavien, Klimaextremereignisse in Frankreich
			\end{itemize}
		\end{column}

\section{Modellvorstellung}
\begin{frame}{PyPSA-Eur: Ein sektorenübergreifendes Energiesystemmodell}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\footnotesize
			\begin{itemize}
				\setlength\itemsep{.8em}
				\item Räumlich und zeitlich hochaufgelöstes lineares Optimierungsmodell, das den \alert{europäischen} Kontinent abdeckt,
				\item Aufbauend auf dem Open-Source-Framework \alert{PyPSA},
				\item Berücksichtigt den \alert{Bestand} bestehender Kraftwerke, erneuerbare Potenziale, \alert{Verfügbarkeitszeitreihen},
				\item Umfasst das \alert{elektrische Übertragungsnetz} von AC 220 kV bis 750 kV (UA) sowie DC ab 150 kV, mit der Option, geplante Netzausbauprojekte (TYNDP und deutscher NEP) einzubeziehen, ebenso das Gasnetz,
				\item Regionale Anwendungen wie z.B. mit PyPSA-DE im Rahmen des \href{https://ariadneprojekt.de/publikation/report-szenarien-zur-klimaneutralitat-2045-2/}{\textbf{Ardiadne-Projekts (Link)}}
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\includegraphics[width=1\textwidth]{pypsa_eur_elec}
		\end{column}
	\end{columns}

	\source{\url{https://pypsa-eur.readthedocs.io}}
\end{frame}

\begin{frame}{Eine typische Modellkonfiguration in PyPSA-Eur \& PyPSA-DE}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\footnotesize
			\begin{itemize}
				\setlength\itemsep{.8em}
				\item \alert{Alle Energiesektoren} (Strom, Wärme, Verkehr, Industrie, Rohstroffe, Landwirtschaft)
				\item \alert{Infrastrukturplanung} von Erzeugung, Übertragung, Speicherung und Power-to-X mit \alert{Technologieprognosen}
				\item \alert{100-200 Regionen} in European
				\item \alert{2-3 stündliche} segmentierte Zeitreihen
				\item \alert{Netto-Null-Emissionen} bis 2050
				\item Minimiere System in \alert{Optimierung}
				\item Viele \alert{Szenarien} durchspielen (oder \alert{stochastische Optimierung})
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\includegraphics[width=1\textwidth]{multisector_diagram}
		\end{column}
	\end{columns}

	\source{\url{https://pypsa-eur.readthedocs.io}}
\end{frame}

						\item H$_2$-Kernnetz weiterhin bereits 2035 verfügbar

						\item Über das Base-Szenario hinaus: Methanolreformierung, Biomethanisierung (Biogasaufbereitung), Biomasse-zu-SNG, Dampfreformierung (SMR)
