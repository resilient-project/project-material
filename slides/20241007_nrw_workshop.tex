\begin{frame}{RESILIENT-Projektübersicht}
	\begin{itemize}
		\item \alert{Problemdiagnose}: Welt befindet sich in einer Phase der \alert{Unsicherheit} und des \alert{Wandels}
		      (Geopolitik, Energiewende, Klimawandel); Szenarien mit einzelnen deterministischen
		      Lösungen nicht mehr bedarfsgerecht
		\item \alert{Projektziel}: Entwicklung und Demonstration von Werkzeugen für eine resiliente Planung der Energiewende für Industrie, E-fuels und Netze
		\item Baut auf dem schon etablierten, quelloffenen und weltweit genutzten Multi-Vektor-Energiesystemmodell \alert{Python for Power System
			      Analysis (PyPSA)}
		\item \alert{Methodischer Schwerpunkt}: Optimierung unter Unsicherheit (stochastisch und robust),
		      Dekompositionsmethoden und Heuristiken für Resilienz
		\item \alert{Fallstudien}: \alert{Industriewende in NRW}, Netzplanung in BW, E-Fuels in
		      Skandinavien, Klimaextremereignisse in Frankreich
		\item Gefördert durch Clean Energy Transition (CET) Partnerschaft (EU-Kofinanzierung)
	\end{itemize}

\end{frame}

\begin{frame}{Fallstudie Industrietransformation in Nordrhein-Westfalen}
	\begin{itemize}
		\item \alert{Problemdiagnose}: Transformation der energieintensiven Industrien erfordert Umstellung von sowohl Brennstoffen als auch Prozessen, Investitionszyklen sind lang, abhängig von verfügbaren Infrastrukturen, viele Unsicherheiten
		\item \alert{RESILIENT-Ansatz}: Herausarbeitung und Konkretisierung der Probleme mit Partnern aus der Industrie $\Rightarrow$ \alert{heutiger Workshop}; Entwicklung von Strategien, die über mehrere Szenarien robust sind
		\item Beispiele für \alert{Unsicherheiten}: Preise und Verfügbarkeit von grünem Wasserstoff; Verfügbarkeit CO$_2$-Pipelines und Biomasse; politische Unterstützung und Marktdynamiken (KSV, grüne Leitmärkte); Akzeptanz für Infrastruktur, CCS, usw.
		\item Beispiele für \alert{Strategien}: Hybridansätze, z.B. bivalente Kessel, Nutzung von (Bio-)Methan für DRI bis H$_2$ da ist; Elektrolyseure vor Ort, CO$_2$-Transport per Schiff/Zug/Lkw, Speicherung von Strom/H$_2$/Derivaten, Flexibilisierung, blauer H$_2$, Import von Zwischenprodukten (HBI, Ammoniak, Methanol)
	\end{itemize}

\end{frame}
