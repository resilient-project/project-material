\begin{frame}{RESILIENT-Projektübersicht}
	\footnotesize
	\begin{columns}
		\begin{column}{0.75\textwidth}
			\begin{itemize}
				\item \alert{Problemdiagnose}: Welt befindet sich in einer Phase der \alert{Unsicherheit} und des \alert{Wandels}
				      (Geopolitik, Energiewende, Klimawandel); Szenarien mit einzelnen deterministischen
				      Lösungen nicht mehr bedarfsgerecht
				\item \alert{Projektziel}: Entwicklung und Demonstration von Werkzeugen für eine resiliente Planung der Energiewende für Industrie, E-fuels und Netze
				\item Baut auf dem schon etablierten, quelloffenen und weltweit genutzten Multi-Vektor-Energiesystemmodell \alert{Python for Power System Analysis (PyPSA)}
				\item \alert{Methodischer Schwerpunkt}: Optimierung unter Unsicherheit (stochastisch und robust),  Dekompositionsmethoden und Heuristiken für Resilienz
				\item \alert{Fallstudien}: \alert{Industriewende in NRW}, Netzplanung in BW, E-Fuels in Skandinavien, Klimaextremereignisse in Frankreich
			\end{itemize}
		\end{column}
		\begin{column}{0.25\textwidth}
			\includegraphics[width=1\textwidth]{logos/logo-secondary-light.png}
		\end{column}
	\end{columns}

\end{frame}

\begin{frame}{Studie: PCI-PMI H$_2$- und CO$_2$-Infrastrukturprojekte}
	\scriptsize
	\begin{columns}[T] % align columns
		% Erste Spalte
		\begin{column}{.63\textwidth}
			\begin{minipage}[t][.45\textheight]{\linewidth}
				\begin{alertblock}{\textbf{Was sind PCI-PMI-Projekte?}}
					\begin{itemize}
						\setlength\itemsep{0.85em}
						\item Projects of Common Interest (PCI) sind zentrale \alert{grenzüberschreitende Infrastrukturprojekte}, die die Energiesysteme der EU-Mitgliedsstaaten verbinden
						\item Ihr Ziel ist es, `der EU beim Erreichen ihrer \alert{energiepolitischen und klimapolitischen Ziele} zu unterstützen: bezahlbare, sichere und nachhaltige Energie für alle Bürgerinnen und Bürger sowie die langfristige Dekarbonisierung der Wirtschaft im Einklang mit dem \alert{Pariser Abkommen}'
						\item Der potenzielle Gesamtnutzen jedes Projekts muss die Kosten übersteigen.
						\item Aufgrund ihres \alert{Leuchtturmcharakters} ist die Umsetzung dieser Projekte sehr wahrscheinlich
						\item Große Infrastrukturprojekte (einschließlich PCI-PMI) sind jedoch häufig von Verzögerungen durch Genehmigungen, Engpässe bei der Beschaffung usw. betroffen
					\end{itemize}
				\end{alertblock}
			\end{minipage}
		\end{column}

		% Second column

				\end{alertblock}

	\end{columns}

\begin{frame}{Übertragungsnetz basierend auf OpenStreetMap (OSM)}

	\begin{columns}
		\begin{column}{0.66\textwidth}
			\footnotesize
			\begin{itemize}
				\setlength\itemsep{.8em}
				\item Datensatz enthält das europäische Hoch- und Höchstspannungsnetz (220 kV bis 750 kV), erstellt auf Basis von OSM-Daten
				\item Ein heuristischer Aufbereitungsprozess wurde angewendet für Einträge, bei denen elektrische Parameter unvollständig, fehlend oder mehrdeutig sind
				\item Umspannwerke innerhalb eines Radius von 500 m werden zu gemeinsamen Knoten aggregiert
				\item Für jedes Spannungspaar in einem Umspannwerk werden Transformatoren hinzugefügt
				\item Wechselstromleitungen werden über Standarddaten der pandapower-Bibliothek parametriert und mit einem Faktor von 0.7 multipliziert, um eine n-1-Sicherheitsnäherung zu berücksichtigen
				\item Enthält alle 38 europäischen HGÜ-Verbindungen in ihrer Nennleistung, die bis Ende 2024 in Betrieb genommen wurden
			\end{itemize}
		\end{column}
		\begin{column}{0.34\textwidth}
			\includegraphics[width=1\textwidth]{osm_map}
		\end{column}
	\end{columns}
	\source{Eigene Darstellung auf Basis von OSM-Daten, extrahiert mit der Overpass turbo API\\\url{https://openstreetmap.org}}

\end{frame}
